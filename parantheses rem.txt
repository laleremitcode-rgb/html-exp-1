#include <stdio.h>
#include <string.h>

#define SIZE 100

char stack[SIZE];
int top = -1;

void push(char c) {
    stack[++top] = c;
}

char pop() {
    if (top == -1) {
        return '\0';
    } else {
        return stack[top--];
    }
}

int main() {
    char exp[SIZE], temp;
    int i;

    printf("Enter an expression: ");
    scanf("%s", exp);

    for (i = 0; exp[i] != '\0'; i++) {
        if (exp[i] == '(' || exp[i] == '{' || exp[i] == '[') {
            push(exp[i]);
        } 
        else if (exp[i] == ')' || exp[i] == '}' || exp[i] == ']') {
            temp = pop();
            if ((exp[i] == ')' && temp != '(') ||
                (exp[i] == '}' && temp != '{') ||
                (exp[i] == ']' && temp != '[')) {
                printf("Not Balanced\n");
                return 0;
            }
        }
    }

    if (top == -1) {
        printf("Balanced\n");
    } 
    else {
        printf("Not Balanced\n");
    }

    return 0;
}
